            // Admin Inquiries panel rendering + Excel CSV export
            const exportBtn = document.getElementById('admin-export-inqs');
            
            function renderAdminInquiries() {
                const inqs = JSON.parse(localStorage.getItem('sm_inquiries')) || [];
                const inqsTableBody = document.getElementById('admin-inqs-table-body');
                
                if (!inqsTableBody) return;
                inqsTableBody.innerHTML = '';
                
                if (inqs.length === 0) {
                    inqsTableBody.innerHTML = '<tr><td colspan="7" style="text-align:center;">No inquiries submitted yet.</td></tr>';
                    return;
                }
                
                inqs.forEach((inq, idx) => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td>${idx + 1}</td>
                        <td><strong>${inq.name}</strong></td>
                        <td>${inq.qualification}</td>
                        <td>${inq.contact}</td>
                        <td>${inq.course}</td>
                        <td>${inq.timestamp}</td>
                        <td>
                            <button class="action-btn-pill btn-delete" onclick="deleteInquiry('${inq.id}')"><i class="fas fa-trash"></i></button>
                        </td>
                    `;
                    inqsTableBody.appendChild(tr);
                });
            }
            
            window.deleteInquiry = function(id) {
                if (!confirm("Are you sure you want to delete this inquiry?")) return;
                let inqs = JSON.parse(localStorage.getItem('sm_inquiries')) || [];
                inqs = inqs.filter(i => i.id !== id);
                localStorage.setItem('sm_inquiries', JSON.stringify(inqs));
                renderAdminInquiries();
                renderAdminStats();
                showToast("Inquiry record deleted.");
            };
            
            if (exportBtn) {
                exportBtn.addEventListener('click', () => {
                    const inqs = JSON.parse(localStorage.getItem('sm_inquiries')) || [];
                    if (inqs.length === 0) {
                        showToast("No inquiries available to export.");
                        return;
                    }
                    
                    // Build CSV Content
                    let csvContent = "data:text/csv;charset=utf-8,";
                    csvContent += "ID,Full Name,Qualification,Address,Contact Number,Email ID,Father Name,Father Contact,Course,Referral Source,Inquiry Date\n";
                    
                    inqs.forEach(inq => {
                        const row = [
                            inq.id,
                            `"${inq.name}"`,
                            `"${inq.qualification}"`,
                            `"${inq.address}"`,
                            `"${inq.contact}"`,
                            `"${inq.email}"`,
                            `"${inq.father}"`,
                            `"${inq.fatherContact}"`,
                            `"${inq.course}"`,
                            `"${inq.source}"`,
                            `"${inq.timestamp}"`
                        ].join(",");
                        csvContent += row + "\n";
                    });
                    
                    // Trigger download link
                    const encodedUri = encodeURI(csvContent);
                    const link = document.createElement("a");
                    link.setAttribute("href", encodedUri);
                    link.setAttribute("download", `STENO_MASTER_inquiries_report_${Date.now()}.csv`);
                    document.body.appendChild(link);
                    link.click();
                    document.body.removeChild(link);
                    showToast("Excel/CSV Report downloaded successfully!");
                });
            }

            // Modal preview outline for blogs
            window.openBlogPreview = function(id) {
                const blogs = JSON.parse(localStorage.getItem('sm_blogs')) || [];
                const blog = blogs.find(b => b.id === id);
                if (!blog) return;
                
                document.getElementById('preview-blog-title').innerText = blog.title;
                document.getElementById('preview-blog-author').innerText = `By: ${blog.author}`;
                document.getElementById('preview-blog-date').innerText = blog.date;
                document.getElementById('preview-blog-content').innerText = blog.content;
                
                const actionsContainer = document.getElementById('preview-modal-actions');
                if (actionsContainer) {
                    actionsContainer.innerHTML = '';
                    if (blog.status === 'pending') {
                        actionsContainer.innerHTML = `
                            <button class="btn btn-primary" style="background-color:#27ae60; border-color:#27ae60; font-size:0.75rem; padding:8px 16px;" onclick="moderateBlog('${blog.id}', 'approve'); closeBlogPreview();"><i class="fas fa-check"></i> Approve & Publish</button>
                            <button class="btn btn-primary" style="background-color:#c0392b; border-color:#c0392b; font-size:0.75rem; padding:8px 16px;" onclick="moderateBlog('${blog.id}', 'reject'); closeBlogPreview();"><i class="fas fa-times"></i> Reject / Delete</button>
                        `;
                    } else {
                        actionsContainer.innerHTML = `
                            <button class="btn btn-primary" style="background-color:#c0392b; border-color:#c0392b; font-size:0.75rem; padding:8px 16px;" onclick="deleteBlog('${blog.id}'); closeBlogPreview();"><i class="fas fa-trash"></i> Delete Post</button>
                            <button class="btn btn-secondary" style="font-size:0.75rem; padding:8px 16px;" onclick="closeBlogPreview()">Close</button>
                        `;
                    }
                }
                
                const modal = document.getElementById('blog-preview-modal');
                if (modal) modal.classList.add('open');
            };
        
            window.closeBlogPreview = function() {

                if (modal) modal.classList.remove('open');

            window.moderateBlog = function(id, action) {
                const blogs = JSON.parse(localStorage.getItem('sm_blogs')) || [];
                const index = blogs.findIndex(b => b.id === id);
                
                if (index !== -1) {
                    blogs[index].status = action === 'approve' ? 'approved' : 'rejected';
                    localStorage.setItem('sm_blogs', JSON.stringify(blogs));
                    
                    renderAdminBlogs();
                    renderAdminApprovedBlogs();
                    renderBlogPosts();
                    renderAdminStats();
                    showToast(`Blog post is now ${blogs[index].status}.`);
                }
            };
            
            window.deleteBlog = function(id) {
                if (!confirm("Are you sure you want to delete this blog post?")) return;
                let blogs = JSON.parse(localStorage.getItem('sm_blogs')) || [];
                blogs = blogs.filter(b => b.id !== id);
                localStorage.setItem('sm_blogs', JSON.stringify(blogs));
                
                renderAdminBlogs();
                renderAdminApprovedBlogs();
                renderBlogPosts();
                renderAdminStats();
                showToast("Blog post deleted.");

            function renderAdminCerts() {
                const certs = JSON.parse(localStorage.getItem('sm_certs')) || [];
                const certsTableBody = document.getElementById('admin-certs-table-body');
                
                if (!certsTableBody) return;
                certsTableBody.innerHTML = '';

                        <td><strong>${c.name}</strong></td>

                        </td>
                        <td>

                    `;
                    certsTableBody.appendChild(tr);
                });

            if (addCertForm) {
                addCertForm.addEventListener('submit', (e) => {
                    e.preventDefault();
                    const id = document.getElementById('admin-c-id').value.trim().toUpperCase();
                    const name = document.getElementById('admin-c-name').value.trim();
                    const course = document.getElementById('admin-c-course').value;
                    const status = document.getElementById('admin-c-status').value;
                    
                    if (!id || !name) {
                        showToast("Please fill in Student ID and Name.");
                        return;
                    }
                    
                    const certs = JSON.parse(localStorage.getItem('sm_certs')) || [];
                    
                    // Check duplicates
                    if (certs.some(c => c.id === id)) {
                        showToast("Error: Student ID already exists.");
                        return;
                    }
                    
                    certs.push({
                        id, name, course, status,
                        date: status === 'completed' ? new Date().toISOString().split('T')[0] : ''
                    });
                    
                    localStorage.setItem('sm_certs', JSON.stringify(certs));
                    renderAdminCerts();
                    renderAdminStats();
                    showToast("Certificate database record created.");
                    addCertForm.reset();
                });
            }

            // Delete a Resource item from catalog
            window.deleteResource = function(category, index) {
                if (!confirm("Are you sure you want to delete this resource item?")) return;
                const res = JSON.parse(localStorage.getItem('sm_resources')) || initialResources;
                
                if (res[category]) {
                    res[category].splice(index, 1);
                    localStorage.setItem('sm_resources', JSON.stringify(res));
                    renderAdminResources();
                    renderResourcesList();
                    showToast("Resource item deleted from catalog.");
                }
            };

                    e.preventDefault();
                    const category = document.getElementById('admin-r-category').value;
