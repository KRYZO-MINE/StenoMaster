                    const inqs = JSON.parse(localStorage.getItem('sm_inquiries')) || [];
                    if (inqs.length === 0) {
                        showToast("No inquiries available to export.");
                        return;
                    }
                    
                    // Build CSV Content
                    let csvContent = "data:text/csv;charset=utf-8,";
                    csvContent += "ID,Full Name,Qualification,Address,Contact Number,Email ID,Father Name,Father Contact,Course,Referral Source,Inquiry Date\n";
                    
                    inqs.forEach(inq => {
                        const row = [
                            inq.id,
                            `"${inq.name}"`,
                            `"${inq.qualification}"`,
                            `"${inq.address}"`,
                            `"${inq.contact}"`,
                            `"${inq.email}"`,
                            `"${inq.father}"`,
                            `"${inq.fatherContact}"`,
                            `"${inq.course}"`,
                            `"${inq.source}"`,
                            `"${inq.timestamp}"`
                        ].join(",");
                        csvContent += row + "\n";
                    });
                    
                    // Trigger download link
                    const encodedUri = encodeURI(csvContent);
                    const link = document.createElement("a");
                    link.setAttribute("href", encodedUri);
                    link.setAttribute("download", `STENO_MASTER_inquiries_report_${Date.now()}.csv`);

            function showToast(message) {
                const container = document.getElementById('toast-container');
                if (!container) return;

                toast.className = 'toast';
                toast.innerHTML = `<i class="fas fa-info-circle"></i> <span>${message}</span>`;
                
                container.appendChild(toast);
                
                // Remove after 4s
                setTimeout(() => {
                    toast.classList.add('hide');
                    setTimeout(() => {
                        toast.remove();
                    }, 400);
                }, 4000);
